\section{Hamiltonsche Formulierung}
\begin{frame}
	\frametitle{Ladungen und Felder}
	\begin{itemize}
		\item Ladungen $Q_{\vec{r}}$ und $Q_{\vec{r}+y}$ an Site $\vec{r}$ und $\vec{r} + y$
		\item kanonisch konjugierte Felder $\hat{E}_{\vec{r}, y}$ und $\hat{A}_{\vec{r}, y}$ am Link $\vec{r}, y$\\
		      $\Rightarrow[\hat{E}_{\vec{r}_j, \mu}, \hat{A}_{\vec{r}_k, \nu}] = i\delta^{(3)}(\vec{r}_j-\vec{r}_k)\delta_{\mu\nu}$
	\end{itemize}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item $\vec{r} + y = (r_x, r_y + 1)$
				\item $\hat{E}_{\vec{r}, \mu}\ket{\hat{e}_{\vec{r}, \mu}}=E_{\vec{r}, \mu}\ket{\hat{e}_{\vec{r}, \mu}}$
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{figure}[h]
				\input{tikz/fl.tex}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}
\begin{frame}
	\frametitle{Link Operator}
	\begin{itemize}
		\item $\hat{U}_{\vec{r}, \mu}=\exp{(iag\hat{A}_{\vec{r}, \mu})}$\\
		      $\Rightarrow\hat{U}_{\vec{r}, \mu}\in$ U(n)
		\item $\hat{A}_{\vec{r}, \mu}$ ist skalar\\
		      $\Rightarrow$ U(1)
		\item Forderung $ag\hat{A}_{\vec{r}, \mu}\in[0,2\pi)$\\
		      $\Rightarrow$ Bild U(1) ist kompakt\\
		      $\Rightarrow$ kompakte U(1) Eichtheorie
		\item $\begin{cases}
				      [\hat{E}_{\vec{r}_{i}, \mu}, \hat{U}_{\vec{r}_{j}, \nu}]      =\delta^{(3)}(\vec{r}_{i}-\vec{r}_{j})\delta_{\mu\nu}\hat{U}_{\vec{r}_{j}, \nu} \\
				      [\hat{E}_{\vec{r}_{i}, \mu}, \hat{U}_{\vec{r}_{j}, \nu}^\dag] =-\delta^{(3)}(\vec{r}_{i}-\vec{r}_{j})\delta_{\mu\nu}\hat{U}_{\vec{r}_{j}, \nu}^{\dag}.
			      \end{cases}$
		\item $\Rightarrow\begin{cases}
				      \hat{U}_{\vec{r}, \mu}\ket{\hat{e}_{\vec{r}, \mu}}=\ket{\hat{e}_{\vec{r}, \mu}-1} \\
				      \hat{U^\dag}_{\vec{r}, \mu}\ket{\hat{e}_{\vec{r}, \mu}}=\ket{\hat{e}_{\vec{r}, \mu}+1}
			      \end{cases}$
	\end{itemize}

\end{frame}

\begin{frame}
	\frametitle{Trunkierung (zweite Diskretisierung)}
	\begin{itemize}
		\item $ag\hat{A}_{\vec{r}, \mu}\in[0,2\pi)\rightarrow ag\hat{A}_{\vec{r}, \mu}\in[-l,l]$ mit Trunkierung $l$
		      \\
		      und $[-l,l]=\{-l,-l+1,\dots,l-1,l\}$\\
		      $\Rightarrow e_{\vec{r}, \mu} \in [-l,l]$
	\end{itemize}
	\begin{columns}
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item nurnoch endlich viele Zustände an einem Link möglich: $N_Z=2l+1$
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			$\hat{U}_{\vec{r}, \mu}=\exp{(iag\hat{A}_{\vec{r}, \mu})}$\\
			\,\\
			U(1) $\subset \mathbb{C}, \{\frac{2\pi k}{2l+1}\}_{k\in\mathbb{Z}_{2l+1}}$\\
			\begin{figure}[h]
				\input{tikz/tunkierung.tex}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}

	mit $\hat{P}_{\vec{r}}=\hat{U}_{\vec{r}, x}\hat{U}_{\vec{r}+x,y}\hat{U}^{\dag}_{\vec{r}+y,x}\hat{U}^{\dag}_{\vec{r},y}$

				      wobei $\ket{j}=\bigotimes_\text{links}\ket{e^{(j)}_{\vec{r}, \mu}}$ und $\mathcal{C}$ der Konfigurationsraum des Gitters ist
			\end{itemize}

\begin{frame}
	\frametitle{Gaußsches Gesetz}
	$\oint_A \vec{E} \cdot \text{d}\vec{A}=\frac{Q}{\epsilon_0} \rightarrow \left[\sum_{\mu\in\{x, y\}}(\hat{E}_{\vec{r}, \mu}-\hat{E}_{\vec{r}-\mu, \mu})-Q_{\vec{r}}\right]\ket{\Psi}=0$

	\begin{columns}
		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item Nur Gitterzustände $\ket{\Psi}$ sind physikalisch: $\ket{\Psi}\in\mathcal{C}_\text{phys}\subset\mathcal{C}$
				\item Gaußsches Gesetz gilt an allen Gitterpunkten $\Rightarrow$ LGS aus $n^2$ Gleichungen
				\item LGS lösen
			\end{itemize}
		\end{column}

\begin{frame}
	\frametitle{Gaußsches Gesetz}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
        \item Links mit frei wählbaren Variablen (E-Feldern: $\hat{E}_{\vec{r}, \text{dyn.}}$) heißen \textit{dynamic links}\\\vspace{10pt}
				$\Rightarrow \hat{E}_{\vec{r}, \text{fixed}}$ sind Linearkombination aus $\hat{E}_{\vec{r}, \text{dyn.}}$\\\vspace{5pt}
				$\Rightarrow \hat{U}_{\vec{r}, \text{fixed}}$ müsse in den Plaquettes nicht mehr beachtet werden, da sie automatisch unphysikalische Gitterzustände erzeugen. Sie würden den Eigenwert von $\hat{E}_{\vec{r}, \text{fixed}}$ ändern, ohne dass dynamische links angepasst werden.
			\end{itemize}
		\end{column}

\begin{frame}
	\frametitle{Beispiel: $2\times2$ ohne Ladung}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Gaußsche Gesetze:
				      \begin{align*}
					      E_{(0, 0), x}+E_{(0,0), y}   & =0 \\
					      E_{(0, 1), y}-E_{(0, 0), x}  & =0 \\
					      -E_{(0, 1), y}-E_{(1, 0), x} & =0 \\
					      E_{(1, 0), x}-E_{(0,0), y}   & =0
				      \end{align*}
				\item Lösung nach $E_{(0, 0), x}$:
				      \begin{align*}
					      E_{(0, 0), y} & =-E_{(0, 0), x} \\
					      E_{(0, 1), y} & =E_{(0, 0), x}  \\
					      E_{(1, 0), x} & =-E_{(0, 0), x}
				      \end{align*}
				\item $\Rightarrow \hat{E}_{(0, 0), x}$ ist dynamisch und nur $\hat{U}_{(0, 0), x}$ kann wirken
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\input{tikz/gauss2.tex}
		\end{column}
	\end{columns}

\end{frame}


\begin{frame}
	\frametitle{Beispiel: $2\times2$ ohne Ladung}
	\begin{itemize}
		\item Trunkierung: $l=1$ $\Rightarrow 2l+1=3$\\
		      $\Rightarrow$ Zustände $\ket{-1}, \ket{0}$ und $\ket{1}$ möglich
		\item $H_{ij}=\bra{i}\hat{H}\ket{j}$ mit $a=1$ und $g=1$\vspace{7pt}\\
		      $ \Rightarrow \hat{H}=
			      \begin{pmatrix}
				      2           & \frac{1}{2} & 0           \\
				      \frac{1}{2} & 0           & \frac{1}{2} \\
				      0           & \frac{1}{2} & 2
			      \end{pmatrix}$\vspace{7pt}\\
		      $\Rightarrow E_0\approx-\num{0.22}$, $E_1=\num{2}$, $E_2\approx\num{2.22}$
        \item $E_0$ am kleinsten $\Rightarrow$ Eigenvektor zu $E_0$ ist Grundzustand $\ket{\Psi_0}$
          \begin{itemize}
            \item kann genutzt werden, um Erwartungswert $\Braket{\hat{O}}=\bra{\Psi_0}\hat{O}\ket{\Psi_0}$ zu berechnen
          \end{itemize}
	\end{itemize}
\end{frame}

\section{Rechenaufwand}
\begin{frame}
	\frametitle{Hamiltonian Größe}
	\begin{itemize}
		\item $N_Z=2l+1$ mit Trunkierung $l$
		\item Anzahl an dynamischen Links: $N_{\text{L,dyn}}$=$N_L-N_{\text{Gleich.}}$
		      \begin{enumerate}
			      \item ohne PBC: $N_L=2n^2-2n$, mit PBC: $N_L=2n^2$\vspace{4pt}
			      \item $\sum_{\vec{r}}\sum_{\mu\in\{x, y\}}(\hat{E}_{\vec{r}, \mu}^2-\hat{E}_{\vec{r}+\mu, \mu}^2)\ket{\Psi}=\sum_{\vec{r}}Q_{\vec{r}}\ket{\Psi}$\vspace{4pt}\\
			            $\Rightarrow$ wenn Gesamtladung Null ist: $N_{\text{Gleich.}}=n^2-1$\vspace{7pt}
		      \end{enumerate}
		\item $N_{\text{phys}}=N_Z^{N_{\text{L, dyn}}}=
			      \begin{cases}
				      (2l+1)^{n^2+1} \text{ mit PBC} \\
				      (2l+1)^{(n-1)^2} \text{ ohne PBC}
			      \end{cases}
		      $\vspace{7pt}
		\item unser Beispiel: $l=1$, $n=2$ ohne PBC $\Rightarrow N_\text{phys}=3$
	\end{itemize}
\end{frame}
